cmake_minimum_required(VERSION 3.25)
project(eos_lhat_tools LANGUAGES C CXX)
set(LHAT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../lhat" CACHE PATH "Matching L^ source checkout")
# Build the language core plus std.binary, which the Bytes overloads need.
set(LHAT_BUILD_TESTS OFF)
set(LHAT_BUILD_CLI OFF)
set(LHAT_BUILD_LSP OFF)
set(LHAT_BUILD_DAP OFF)
set(LHAT_BUILD_STDLIB ON)
set(LHAT_BUILD_STDLIB_NET OFF)
set(LHAT_BUILD_BENCH OFF)
add_subdirectory("${LHAT_ROOT}" lhat EXCLUDE_FROM_ALL)
add_executable(eos_lhat_host host.cpp)
target_compile_features(eos_lhat_host PRIVATE cxx_std_17)
target_include_directories(eos_lhat_host PRIVATE "${LHAT_ROOT}")
target_link_libraries(eos_lhat_host PRIVATE lhatstdlib lhat ${CMAKE_DL_LIBS})
