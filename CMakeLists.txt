cmake_minimum_required(VERSION 3.25)
project(eos_lhat VERSION 0.1.0 LANGUAGES C CXX)
find_package(Threads REQUIRED)
set(EOS_SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/EOSSDK/SDK" CACHE PATH "Downloaded EOS SDK (Include, Lib, Bin)")
set(LHAT_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../lhat" CACHE PATH "Matching L^ source checkout")
add_subdirectory(tools EXCLUDE_FROM_ALL)
set(LHAT_GENERATED_INCLUDE "${CMAKE_CURRENT_BINARY_DIR}/tools/lhat/include")
if(NOT EXISTS "${EOS_SDK_ROOT}/Include/eos_sdk.h")
    message(FATAL_ERROR "Download the EOS SDK separately and set EOS_SDK_ROOT to its SDK directory")
endif()
if(WIN32)
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
        message(FATAL_ERROR "The desktop EOS SDK requires a 64-bit build")
    endif()
    set(EOS_LIBRARY_NAME EOSSDK-Win64-Shipping)
    if(CMAKE_GENERATOR_PLATFORM MATCHES "[Aa][Rr][Mm]64" OR CMAKE_SYSTEM_PROCESSOR MATCHES "[Aa][Rr][Mm]64|aarch64")
        set(EOS_LIBRARY_NAME EOSSDK-Win64-Shippingarm64)
    endif()
elseif(APPLE)
    set(EOS_LIBRARY_NAME EOSSDK-Mac-Shipping)
else()
    set(EOS_LIBRARY_NAME EOSSDK-Linux-Shipping)
endif()
find_library(EOS_LIBRARY NAMES ${EOS_LIBRARY_NAME} PATHS "${EOS_SDK_ROOT}/Lib" "${EOS_SDK_ROOT}/Bin" NO_DEFAULT_PATH REQUIRED)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")
if(NOT EXISTS "${CMAKE_CURRENT_BINARY_DIR}/generated/eos_signatures.h")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/generated/eos_signatures.h" "static const unsigned char eos_signatures[] = {0};\nstatic const size_t eos_signatures_length = 0;\n")
endif()
add_library(eos_lhat SHARED src/client.cpp src/bindings.cpp)
add_dependencies(eos_lhat eos_lhat_host)
target_compile_features(eos_lhat PRIVATE cxx_std_17)
target_include_directories(eos_lhat PRIVATE "${EOS_SDK_ROOT}/Include" "${LHAT_ROOT}" "${LHAT_ROOT}/include" "${LHAT_GENERATED_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(eos_lhat PRIVATE "${EOS_LIBRARY}" Threads::Threads)
set_target_properties(eos_lhat PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN ON)
if(MSVC)
    target_compile_options(eos_lhat PRIVATE /W4 /EHsc)
else()
    target_compile_options(eos_lhat PRIVATE -Wall -Wextra -Wpedantic)
    set_target_properties(eos_lhat PROPERTIES BUILD_RPATH "${EOS_SDK_ROOT}/Bin")
    if(APPLE)
        # Link with the final path so installation need not edit the signed Mach-O.
        set_target_properties(eos_lhat PROPERTIES INSTALL_RPATH "@loader_path" BUILD_WITH_INSTALL_RPATH TRUE)
        # The EOS SDK dylib is universal; ship one extension for both Mac CPUs.
        option(EOS_LHAT_MACOS_UNIVERSAL "Build eos_lhat for arm64 and x86_64" ON)
        if(EOS_LHAT_MACOS_UNIVERSAL)
            set_target_properties(eos_lhat PROPERTIES OSX_ARCHITECTURES "arm64;x86_64")
        endif()
    else()
        set_target_properties(eos_lhat PROPERTIES INSTALL_RPATH "$ORIGIN")
    endif()
endif()
# Neither proprietary SDK binaries nor a separate lhat runtime are bundled here.
install(TARGETS eos_lhat RUNTIME DESTINATION native LIBRARY DESTINATION native)
include(CTest)
if(BUILD_TESTING)
    add_executable(test_client src/client.cpp tests/fake_eos.cpp tests/test_client.cpp)
    target_compile_features(test_client PRIVATE cxx_std_17)
    target_link_libraries(test_client PRIVATE Threads::Threads)
    target_compile_definitions(test_client PRIVATE EOS_MONOLITHIC=1)
    target_include_directories(test_client PRIVATE src "${EOS_SDK_ROOT}/Include")
    add_test(NAME client COMMAND test_client)
    add_library(eos_lhat_mock SHARED src/client.cpp src/bindings.cpp tests/fake_eos.cpp)
    target_compile_features(eos_lhat_mock PRIVATE cxx_std_17)
    target_link_libraries(eos_lhat_mock PRIVATE Threads::Threads)
    target_compile_definitions(eos_lhat_mock PRIVATE EOS_MONOLITHIC=1)
    target_include_directories(eos_lhat_mock PRIVATE src "${EOS_SDK_ROOT}/Include" "${LHAT_ROOT}" "${LHAT_ROOT}/include" "${LHAT_GENERATED_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}/generated")
    set_target_properties(eos_lhat_mock PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
endif()
